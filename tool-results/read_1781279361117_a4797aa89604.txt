   287→   287→// ─── Badge Component ────────────────────────────────────────────────────────
   288→   288→
   289→   289→function Badge({ type, text }: { type: BadgeType; text?: string }) {
   290→   290→  if (!type) return null;
   291→   291→  const classes =
   292→   292→    type === 'TOP'
   293→   293→      ? 'bg-[#FF3366] text-white'
   294→   294→      : type === 'NEW'
   295→   295→        ? 'bg-[#D7FF00] text-black'
   296→   296→        : type === 'SOON'
   297→   297→          ? 'bg-white/10 text-white/70 border border-white/10'
   298→   298→          : 'bg-[#FF3366] text-white';
   299→   299→
   300→   300→  return (
   301→   301→    <span
   302→   302→      className={`${classes} text-[9px] font-bold px-1.5 py-0.5 rounded leading-none tracking-wider uppercase shrink-0`}
   303→   303→    >
   304→   304→      {text || type}
   305→   305→    </span>
   306→   306→  );
   307→   307→}
